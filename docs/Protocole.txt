Les deux premières bases CC appartiennent à un codon incomplet au début du fragment. Le premier codon complet commence donc à la troisième base : c’est exactement offset 2 dans notre script. Tu peux extraire directement la liste des hits de ton rapport BLAST et créer un fichier avec les colonnes dans l’ordre attendu :

Ce bloc lit ton rapport, extrait les accessions sans doublons et écrit un fichier intermédiaire, sans écraser references/H1.tsv :

python - <<'PY'
import re
from pathlib import Path

folder = Path("analyses/H1/03_blast")
source = folder / "H1_BLAST_2026-11-10_F5XCKAV5016_report.txt"
destination = folder / "H1_references_depuis_BLAST.tsv"

rows = []
seen = set()
in_table = False

for line in source.read_text().splitlines():
    if "Sequences producing significant alignments:" in line:
        in_table = True
        continue

    if not in_table:
        continue

    if line.lstrip().startswith(">"):
        break

    if re.match(r"^\s*(?:Alignments\b|Score\s*=|Query\s+\d+)", line):
        break

    match = re.search(r"\b([A-Z]{1,6}_?\d+\.\d+)\s*$", line)
    if not match:
        continue

    accession = match.group(1)
    if accession in seen:
        continue

    name = re.match(r"^\s*([A-Z][a-z]+)\s+([a-z][a-z-]+)\b", line)
    species = " ".join(name.groups()) if name else ""

    rows.append("\t".join([
        accession,
        species,
        "reference",
        "BLAST H1 ; RID F5XCKAV5016 ; nom a verifier"
    ]))
    seen.add(accession)

if not rows:
    raise SystemExit("Aucune accession extraite : verifier le rapport.")

header = "accession\texpected_species\trole\tnotes"
destination.write_text(header + "\n" + "\n".join(rows) + "\n")

print(f"{len(rows)} references extraites.")
print(f"Fichier : {destination}")
PY
